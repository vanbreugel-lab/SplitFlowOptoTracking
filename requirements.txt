# --- Core scientific stack ---
numpy
# IMPORTANT: pandas must be < 3.0.  pandas 3.0 makes Copy-on-Write mandatory,
# which returns read-only arrays from DataFrame.values; the unifying-algorithm
# alignment code (align_course_direction_analysis) does in-place array ops and
# raises "output array is read-only" under pandas >= 3.0. Pin to the 2.2.x line.
pandas<3.0
scipy
matplotlib
scikit-learn        # sklearn.preprocessing (autoencoder, several figures)
seaborn             # fig_4 autoencoder plots
cvxpy               # unifying-algorithm mixed-integer quadratic program

# --- Deep learning (fig_4 autoencoder / triplet VAE) ---
torch
optuna              # hyperparameter search notebook (fig_4/misc)

# --- Optimization solver (REQUIRED for the unifying algorithm) ---
# The unifying-algo fit is a MIQP solved with solver=cvxpy.GUROBI.
# Gurobi is commercial but free for academics (https://www.gurobi.com/academia/).
gurobipy

# --- I/O backends (used at runtime, not imported by name) ---

tables              # pandas .read_hdf (*.hdf); PyTables

cairosvg            # unifying_simulation/plot_simulation.ipynb (SVG -> PNG frames)
